### 3b. `performDeduction` — API 调用

```swift
func performDeduction(question, background, liuYaoChart, castTime, canonicalSummary) async throws -> DeductionReport {
    let prompt = buildDeductionPrompt(...)
    let requestBody = [
        "model": ConfigManager.shared.modelEndpoint,
        "messages": [["role": "user", "content": prompt]],
        "max_tokens": 8000,          // 推演 JSON 体积大，与 master 模式相同
        "temperature": 0.5
    ]
    let response = try await NetworkService.shared.sendRequest(body: requestBody, responseType: AIResponse.self)
    guard let content = response.choices.first?.message.content else { throw AIServiceError.noResponse }
    return try parseDeductionReport(from: content)
}
```

### 3c. `parseDeductionReport` — JSON 提取+解码

AI 可能在 JSON 外包 markdown 代码块，需先提取纯 JSON：
```swift
// 优先级：```json\n...\n``` → {...} 首尾花括号裁剪
private func parseDeductionReport(from content: String) throws -> DeductionReport {
    var json = content
    if let s = content.range(of: "```json\n"), let e = content.range(of: "\n```", range: s.upperBound..<content.endIndex) {
        json = String(content[s.upperBound..<e.lowerBound])
    } else if let s = content.firstIndex(of: "{"), let e = content.lastIndex(of: "}") {
        json = String(content[s...e])
    }
    return try JSONDecoder().decode(DeductionReport.self, from: Data(json.utf8))
}
```

```swift
private func startDeduction() {
    guard let chart = liuYaoChart else { return }
    isLoading = true
    Task {
        do {
            let parsed = MasterReportParser.parse(aiInterpretation)
            let summary = buildCanonicalSummary(from: parsed)
            let report = try await AIService.shared.performDeduction(
                question: derivedQuestion, background: backgroundText,
                liuYaoChart: chart, castTime: castTime, canonicalSummary: summary)
            await MainActor.run { deductionReport = report; isLoading = false; showDeductionResult = true }
        } catch {
            await MainActor.run { isLoading = false; errorMessage = "推演失败，请重试" }
        }
    }
}
```
